function desenhar(id, type, labels, data, label) { if (charts[id]) charts[id].destroy(); const cores = ['#e74c3c', '#3498db', '#f1c40f', '#2ecc71']; // Define a escala: SAP trava no 5, DISC fica livre para mostrar a diferença let configuracaoEscala = { beginAtZero: true }; if (id.includes('Sap')) { configuracaoEscala.max = 5; // Trava o SAP em 5 } charts[id] = new Chart(document.getElementById(id).getContext('2d'), { type: type, data: { labels: labels, datasets: [{ label: label, data: data, backgroundColor: id.includes('Disc') ? cores : 'rgba(0, 86, 179, 0.4)', borderColor: '#0056b3', borderWidth: 1 }] }, options: { responsive: true, maintainAspectRatio: false, scales: (type === 'bar' || type === 'radar') ? { y: configuracaoEscala, r: id.includes('Sap') ? { beginAtZero: true, max: 5 } : { beginAtZero: true } } : {} } }); }
